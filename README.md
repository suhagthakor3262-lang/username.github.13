    function filterApps() {
        var input = document.getElementById('vipSearchField').value.toLowerCase();
        var items = document.getElementsByClassName('search-item');
        
        for (var i = 0; i < items.length; i++) {

            if (name) {
                if (name.includes(input)) {
                    if (items[i].classList.contains('vip-game-card')) {
                        items[i].style.display = "flex";
                    } else {
                        items[i].style.display = "";
                    }
                } else {
                    items[i].style.display = "none";
                }
            }
        }
    }

    // 2. Interactive Legal Tabs Engine
    function toggleLegalTab(tabKey) {
        var contents = document.getElementsByClassName('vip-legal-content');
        for (var i = 0; i < contents.length; i++) {
            contents[i].style.display = "none";
        }
        var targetNode = document.getElementById('legal-' + tabKey);
        if (targetNode) {
            targetNode.style.display = "block";
            targetNode.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }
    }

    // 3. Persistent 13 Days Background Core Logic
    (function() {
        var duration = (13 * 24 * 60 * 60) + (18 * 60 * 60) + (41 * 60); 
        var storageKey = "winRummyTargetTimeV2";
        var targetTime = localStorage.getItem(storageKey);

        if (!targetTime) {
            targetTime = new Date().getTime() + (duration * 1000);
            localStorage.setItem(storageKey, targetTime);
        } else {
            targetTime = parseInt(targetTime, 10);
        }

        var timerInterval = setInterval(function() {
            var now = new Date().getTime();
            var timeRemaining = targetTime - now;

            if (timeRemaining <= 0) {
                clearInterval(timerInterval);
                document.getElementById("winRummyTimerWidget").style.display = "none";
                document.getElementById("winRummyCtaBlock").style.display = "flex";
                return;
            }

            var days = Math.floor(timeRemaining / (1000 * 60 * 60 * 24));
            var hours = Math.floor((timeRemaining % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            var minutes = Math.floor((timeRemaining % (1000 * 60 * 60)) / (1000 * 60));
            var seconds = Math.floor((timeRemaining % (1000 * 60)) / 1000);

            document.getElementById("timer-days").innerText = String(days).padStart(2, '0');
            document.getElementById("timer-hours").innerText = String(hours).padStart(2, '0');
            document.getElementById("timer-mins").innerText = String(minutes).padStart(2, '0');
            document.getElementById("timer-secs").innerText = String(seconds).padStart(2, '0');
        }, 1000);
    })();
